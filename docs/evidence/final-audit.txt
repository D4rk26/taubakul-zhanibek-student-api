## Secrets: search the whole history (git log --all -p) for the real secret values
(values are read from the secret files inside the command and never printed)

$ for name in jenkins-admin-password jenkins-api-token db-password github-token; do
  case $name in
    jenkins-admin-password) v=$(cat ~/projects/jenkins-admin-password.txt) ;;
    jenkins-api-token)      v=$(cut -d: -f2 ~/projects/jenkins-api-token.txt) ;;
    db-password)            v=$(grep "^DB_PASSWORD=" .env | cut -d= -f2-) ;;
    github-token)           v=$(gh auth token) ;;
  esac
  [ ${#v} -ge 20 ] || { echo "$name: value not loaded"; continue; }
  n=$(git log --all -p | grep -cF "$v"); m=$(grep -rlF "$v" . --exclude-dir=.git --exclude-dir=build --exclude-dir=.gradle --exclude=.env | wc -l | tr -d " ")
  echo "$name: matches in history=$n, in working tree files=$m"
done; unset v
